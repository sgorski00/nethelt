Table users {
  id bigint [pk, increment]
  email varchar(255) [not null, note: 'unique among not deleted rows']
  password_hash varchar(255) [note: 'BCrypt, null for OAuth2-only accounts']
  role varchar(50) [not null, note: 'USER, ADMIN']
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft delete']
}

Table profiles {
  id bigint [pk, increment]
  user_id bigint [not null, unique]
  username varchar(255) [not null, unique]
  first_name varchar(255)
  last_name varchar(255)
  birth_date date
  bio text
  created_at timestamp
  updated_at timestamp
}

Table user_identities {
  id bigint [pk, increment]
  user_id bigint [not null]
  provider varchar(255) [not null, note: 'GOOGLE, GITHUB']
  provider_id text [not null]

  indexes {
    (provider, provider_id) [unique]
    (user_id, provider) [unique]
  }
}

Table refresh_tokens {
  id bigint [pk, increment]
  token varchar(255) [not null, unique]
  user_id bigint [not null]
  is_revoked boolean [not null, default: false]
  expires_at timestamp [not null]
}

Table password_reset_tokens {
  id bigint [pk, increment]
  token varchar(255) [not null, unique]
  user_id bigint [not null]
  is_revoked boolean [not null, default: false]
  expires_at timestamp [not null]
}

Table networks {
  id bigint [pk, increment]
  user_id bigint [not null]
  name varchar(255) [not null, note: 'unique per user among not deleted rows']
  description text
  created_at timestamp
  updated_at timestamp
  deleted_at timestamp [note: 'soft delete']
}

Table network_agents {
  id bigint [pk, increment]
  network_id bigint [not null, unique]
  name varchar(255) [not null]
  hashed_token text [not null, unique]
  token_created_at timestamp [not null]
  last_heartbeat_at timestamp
  status varchar(50) [not null, default: 'ACTIVE']
  max_unavailability_time interval [not null, default: '5 minutes']
  availability_alert_state varchar(16) [not null, default: 'OK']
}

Table devices {
  id bigint [pk, increment]
  network_id bigint [not null]
  name varchar(255) [not null]
  ip_address inet [not null, note: 'IPv4 only']
  ip_numeric bigint [not null, note: 'IP as number, used for range filtering']
  type varchar(50) [not null, note: 'LAN_CLIENT, WIFI_CLIENT, NETWORK_DEVICE']
  is_enabled boolean [not null, default: true]
  created_at timestamptz [not null]
  updated_at timestamptz [not null]

  indexes {
    (network_id, name) [unique]
    (network_id, ip_address) [unique]
    (network_id, ip_numeric)
  }
}

Table monitoring_tasks {
  id bigint [pk, increment]
  device_id bigint [not null]
  configuration_id bigint
  type varchar(255) [not null, note: 'PING, TELNET, HTTP_HEALTHCHECK']
  interval interval [not null, note: 'delay between runs']
  is_enabled boolean [not null, default: true]
  is_ai_analysis_enabled boolean [not null, default: false]
  created_at timestamptz [not null]
  updated_at timestamptz [not null]
}

}

  id bigint [pk, increment]
  task_id bigint [not null]
  monitoring_result_extension_id bigint [not null, unique]

  success boolean [not null]
  message text
  response_time_ms bigint
  created_at timestamptz [not null]

  indexes {

Table ai_analysis_results {
  id bigint [pk, increment]
  task_id bigint [not null]
  window_start timestamptz [not null, note: '1-minute window']
  ping_std_1m double
  packet_loss_1m double
  score double [not null, note: 'Isolation Forest score, below 0 = anomaly']
  is_anomaly boolean [not null]
  model_version varchar(50) [not null]
  created_at timestamptz [not null]

  indexes {
    (task_id, window_start) [unique]
  }
}
